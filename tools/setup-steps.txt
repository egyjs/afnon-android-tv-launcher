# What Afnon needs from the TV after `adb install`. The one source for both tools/install.sh and the phone
# installer, which reads this file from the release's tag, so a new step here reaches every installer without
# updating it. Keep commands one line, safe to run again, and harmless on brands without the setting.
#
#   required|English label|Arabic label|command
#   ?|check
#
# The check (optional, on the line after its step) exits 0 when the step is in place on the TV. Both tools run
# it first and skip a step that already passes, and run it again after the command: these are the settings a
# firmware update, a reboot or the TV's own housekeeping can undo, so "set" is not the same as "checked".
# Home: the HOME role and set-home-activity are enough on most TVs, but a stock Google launcher wins over any app
# (tvlauncher's filter has priority 2; Google TV's launcherx, which TCL also starts by name). Then the launcher
# (plus Google's setup app on Google TV) is disabled for the TV's user and noted in afnon-cleaned.txt for undo,
# behind a safety switch on the TV that turns it back on after 5 minutes unless Afnon came up and stayed up
# (the same process 3 s later: a crash-looping Home also looks "resumed"). If it didn't, it is put back at once.
1|Afnon is the home screen|أفنون هو الشاشة الرئيسية|O=/data/local/tmp/afnon_home_ok; C=/data/local/tmp/afnon-cleaned.txt; on() { pm list packages -e "$1" | grep -qx "package:$1"; }; front() { a=$(pidof tv.afnon.home) || return 1; sleep 3; dumpsys activity activities | grep -m1 -E 'topResumedActivity|mResumedActivity' | grep -q ' tv.afnon.home/' && [ "$(pidof tv.afnon.home)" = "$a" ]; }; home() { cmd package set-home-activity --user 0 tv.afnon.home/.MainActivity >/dev/null; for i in 1 2 3; do input keyevent KEYCODE_HOME; sleep 2; front && return 0; done; return 1; }; cmd role add-role-holder --user 0 android.app.role.HOME tv.afnon.home >/dev/null 2>&1; X=com.google.android.apps.tv.launcherx; T=com.google.android.tvlauncher; if on $X; then S=$X/$X.home.HomeActivity; P="com.google.android.tungsten.setupwraith $X"; elif on $T; then S=$T/.MainActivity; P=$T; else P=; fi; H=$(cmd package resolve-activity --brief -a android.intent.action.MAIN -c android.intent.category.HOME | tail -n 1); case "$H" in "${P##* }"/*) S=$H ;; esac; if [ -z "$P" ]; then home; else rm -f $O; nohup sh -c "i=0; while [ \$i -lt 300 ]; do [ -f $O ] && exit 0; sleep 1; i=\$((i+1)); done; for p in $P; do pm enable \$p; done; cmd package set-home-activity --user 0 $S; input keyevent KEYCODE_HOME" >/dev/null 2>&1 & for p in $P; do on $p && pm disable-user --user 0 $p >/dev/null && { grep -qx $p $C 2>/dev/null || echo $p >> $C; }; done; if home; then touch $O; else for p in $P; do pm enable $p >/dev/null; done; cmd package set-home-activity --user 0 $S >/dev/null; input keyevent KEYCODE_HOME; touch $O; false; fi; fi
?|cmd package resolve-activity --brief -a android.intent.action.MAIN -c android.intent.category.HOME | tail -n 1 | grep -q '^tv.afnon.home/' && ! pm list packages -e com.google.android.apps.tv.launcherx | grep -q .
1|Remote buttons and the call to prayer|أزرار جهاز التحكم والأذان|s=$(settings get secure enabled_accessibility_services); case "$s" in *tv.afnon.home/.RemoteKeys*) ;; ''|null) s=tv.afnon.home/.RemoteKeys ;; *) s="$s:tv.afnon.home/.RemoteKeys" ;; esac; settings put secure enabled_accessibility_services "$s" && settings put secure accessibility_enabled 1
?|settings get secure enabled_accessibility_services | grep -q 'tv.afnon.home/.RemoteKeys' && [ "$(settings get secure accessibility_enabled)" = 1 ]

?|settings get secure screensaver_components | grep -q 'tv.afnon.home/.AfnonDream'

?|o=$(appops get tv.afnon.home APP_AUTO_START 2>&1); case "$o" in *allow*) true ;; *) false ;; esac
